import React from 'react';
import { WESTCLIFF_THEME } from './theme';

export const SlideRenderer = ({ slide }) => {
  const { colors, typography } = WESTCLIFF_THEME;

  const containerStyle = {
    backgroundColor: slide.type === 'SECTION_HEADER' ? colors.primaryNavy : colors.bgLight,
    color: slide.type === 'SECTION_HEADER' ? '#FFFFFF' : colors.textDark,
    fontFamily: typography.fontFamilyBody,
    padding: '2.5rem 3.5rem',
    minHeight: '100vh',
    display: 'flex',
    flexDirection: 'column',
    justifyContent: 'space-between',
    boxSizing: 'border-box'
  };

  const headerStyle = {
    borderBottom: slide.type === 'SECTION_HEADER' ? 'none' : `2px solid ${colors.accentGold}`,
    paddingBottom: '0.75rem',
    marginBottom: '1.5rem'
  };

  const titleStyle = {
    fontFamily: typography.fontFamilyTitle,
    fontSize: typography.sizes.title,
    color: slide.type === 'SECTION_HEADER' ? colors.accentGold : colors.primaryNavy,
    margin: 0
  };

  const chapterTagStyle = {
    fontSize: typography.sizes.caption,
    fontWeight: typography.weights.bold,
    color: colors.accentGold,
    textTransform: 'uppercase',
    letterSpacing: '1px',
    marginBottom: '0.25rem'
  };

  return (
    <div style={containerStyle}>
      {/* Slide Top Header */}
      {slide.type !== 'SECTION_HEADER' && slide.type !== 'TITLE_SLIDE' && (
        <div style={headerStyle}>
          {slide.chapter && <div style={chapterTagStyle}>{slide.chapter}</div>}
          <h1 style={titleStyle}>{slide.title}</h1>
          {slide.subtitle && <p style={{ color: colors.textMuted, margin: '0.25rem 0 0 0' }}>{slide.subtitle}</p>}
        </div>
      )}

      {/* Main Content Body */}
      <div style={{ flex: 1, display: 'flex', flexDirection: 'column', justifyContent: 'center' }}>
        {/* Title Slide */}
        {slide.type === 'TITLE_SLIDE' && (
          <div style={{ textAlign: 'center', maxWidth: '900px', margin: '0 auto' }}>
            <h3 style={{ color: colors.accentGold, letterSpacing: '2px' }}>{slide.meta.institution}</h3>
            <h1 style={{ fontFamily: typography.fontFamilyTitle, fontSize: '2.75rem', color: colors.primaryNavy, margin: '1rem 0' }}>
              {slide.title}
            </h1>
            <h2 style={{ fontSize: typography.sizes.subtitle, color: colors.secondaryBlue, fontWeight: typography.weights.regular }}>
              {slide.subtitle}
            </h2>
            <div style={{ marginTop: '3rem', borderTop: `1px solid ${colors.border}`, paddingTop: '1.5rem' }}>
              <p style={{ fontWeight: typography.weights.bold, margin: 0 }}>{slide.meta.author} · {slide.meta.degree}</p>
              <p style={{ color: colors.textMuted, margin: '0.25rem 0' }}>{slide.meta.defenseType} | {slide.meta.date}</p>
            </div>
          </div>
        )}

        {/* Section Headers */}
        {slide.type === 'SECTION_HEADER' && (
          <div style={{ textAlign: 'center' }}>
            <p style={{ color: colors.accentGold, fontSize: '1.25rem', letterSpacing: '2px' }}>{slide.chapter}</p>
            <h1 style={{ fontFamily: typography.fontFamilyTitle, fontSize: '3.25rem', color: '#FFFFFF', margin: '0.5rem 0' }}>
              {slide.title}
            </h1>
            {slide.subtitle && <p style={{ color: '#E2E8F0', fontSize: '1.25rem' }}>{slide.subtitle}</p>}
          </div>
        )}

        {/* Bullet Card Lists */}
        {slide.type === 'CONTENT_CARD_LIST' && (
          <div style={{ display: 'flex', flexDirection: 'column', gap: '1rem' }}>
            {slide.bullets.map((bullet, idx) => (
              <div 
                key={idx} 
                style={{ 
                  backgroundColor: colors.cardBg, 
                  padding: '1.25rem 1.5rem', 
                  borderRadius: '8px', 
                  borderLeft: `4px solid ${colors.accentGold}`,
                  boxShadow: '0 2px 4px rgba(0,0,0,0.04)',
                  lineHeight: typography.lineHeight.relaxed
                }}
              >
                {bullet}
              </div>
            ))}
          </div>
        )}

        {/* Grid 3 / Grid 4 Layouts */}
        {(slide.type === 'GRID_3_CARDS' || slide.type === 'GRID_4_BOXES') && (
          <div style={{ display: 'grid', gridTemplateColumns: slide.items.length === 4 ? '1fr 1fr' : '1fr 1fr 1fr', gap: '1.25rem' }}>
            {slide.items.map((item, idx) => (
              <div key={idx} style={{ backgroundColor: colors.cardBg, padding: '1.5rem', borderRadius: '8px', border: `1px solid ${colors.border}` }}>
                <h3 style={{ color: colors.primaryNavy, marginTop: 0, borderBottom: `2px solid ${colors.accentGold}`, paddingBottom: '0.5rem' }}>
                  {item.title}
                </h3>
                {item.author && <p style={{ fontSize: typography.sizes.caption, color: colors.accentGold, fontWeight: 'bold' }}>{item.author}</p>}
                <p style={{ lineHeight: typography.lineHeight.relaxed, fontSize: '0.95rem' }}>{item.text || item.description}</p>
              </div>
            ))}
          </div>
        )}

        {/* Two Column Layout */}
        {slide.type === 'TWO_COLUMN' && (
          <div style={{ display: 'grid', gridTemplateColumns: '1fr 1fr', gap: '2rem' }}>
            {[slide.leftColumn, slide.rightColumn].map((col, idx) => (
              <div key={idx} style={{ backgroundColor: colors.cardBg, padding: '1.5rem', borderRadius: '8px', borderTop: `4px solid ${colors.secondaryBlue}` }}>
                <h3 style={{ color: colors.primaryNavy, marginTop: 0 }}>{col.heading}</h3>
                <ul style={{ paddingLeft: '1.2rem', lineHeight: typography.lineHeight.relaxed }}>
                  {col.bullets.map((b, bIdx) => (
                    <li key={bIdx} style={{ marginBottom: '0.75rem' }}>{b}</li>
                  ))}
                </ul>
              </div>
            ))}
          </div>
        )}
      </div>

      {/* Slide Footer */}
      {slide.type !== 'TITLE_SLIDE' && (
        <div style={{ display: 'flex', justifyContent: 'space-between', fontSize: typography.sizes.caption, color: colors.textMuted, borderTop: `1px solid ${colors.border}`, paddingTop: '0.75rem', marginTop: '1.5rem' }}>
          <span>WESTCLIFF UNIVERSITY · Doctoral Defense</span>
          <span>Slide {slide.slideNumber} of 34</span>
        </div>
      )}
    </div>
  );
};
